```bash
# 1) 起栈前静态体检（含 config/map/路线；不依赖节点在跑）
bash ~/dzrobot/tools/healthcheck.sh --pre
# 2) 起全栈：默认新核 dzbrain；dzbrain 未安装会自动回退旧核
bash ~/dzrobot/start_all.sh
LEGACY_PATH=1 bash ~/dzrobot/start_all.sh          # 一键回退旧循迹核（path_executor）

tail -f <LOGDIR>/7_path.log      # 循迹（新核 dzbrain 输出）

# 5) 人接管 / 让车停（std_msgs/UInt8，1=接管，0=交回自动；可恢复，不是安全锁）
ros2 topic pub --once /nav/manual_override std_msgs/msg/UInt8 "{data: 1}"
#    语义（已核实）：safety.py:201-210 只记录"接管中"、安全判据继续生效；
#    tracker.py:260-271 前置门命中 ⇒ publish_brake() **循迹立即刹停并保持**；
#    aim.py:855 硬门 2 ⇒ 云台一并让出。发 0 即交回自动。

```

> ⚠️ 注意：`/nav/manual_override` 是"**接管让位**"（可恢复），**不等于** SAFE_STOP 那种自锁；要彻底停止请用 `start_all.sh stop`。
